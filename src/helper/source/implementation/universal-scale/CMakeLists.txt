cmake_minimum_required(VERSION 3.24)
project(nr_universal_scale LANGUAGES CXX)
if(NOT TARGET nr_compat)
  add_subdirectory(../compat-diag compat-diag)
endif()
add_library(nr_ffx_descriptor STATIC ../active-scheduling/ffx_descriptor.cpp ../active-scheduling/legacy_fsr3_descriptor.cpp)
target_compile_features(nr_ffx_descriptor PUBLIC cxx_std_17)
target_include_directories(nr_ffx_descriptor PUBLIC ../active-scheduling)
target_compile_options(nr_ffx_descriptor PRIVATE /W4 /WX /permissive- /EHsc)
add_library(nr_scale_policy STATIC scale_policy.cpp dispatch_flow.cpp coordinator.cpp input_adapter.cpp)
target_compile_features(nr_scale_policy PUBLIC cxx_std_17)
target_include_directories(nr_scale_policy PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ../active-scheduling)
target_link_libraries(nr_scale_policy PUBLIC nr_ffx_descriptor nr_compat)
target_compile_options(nr_scale_policy PRIVATE /W4 /WX /permissive- /EHsc)

if(NOT NR_SCALE_DXC)
  set(NR_SCALE_DXC "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../toolchains/rga-2.14.2/utils/dx12/dxc/dxc.exe")
endif()
if(NOT EXISTS "${NR_SCALE_DXC}")
  message(FATAL_ERROR "Pinned DXC missing")
endif()
set(NR_SCALE_SHADER_DIR "${CMAKE_CURRENT_BINARY_DIR}/shaders")
file(MAKE_DIRECTORY "${NR_SCALE_SHADER_DIR}")
foreach(entry IN ITEMS ColorCS MotionCS DepthCS)
  string(TOLOWER "${entry}" lower)
  add_custom_command(OUTPUT "${NR_SCALE_SHADER_DIR}/${lower}.cso"
    COMMAND "${NR_SCALE_DXC}" -E "${entry}" -T cs_6_0 -O3 -Ges -Fo "${NR_SCALE_SHADER_DIR}/${lower}.cso" "${CMAKE_CURRENT_SOURCE_DIR}/nr_scale.hlsl"
    DEPENDS nr_scale.hlsl VERBATIM)
  list(APPEND NR_SCALE_SHADERS "${NR_SCALE_SHADER_DIR}/${lower}.cso")
endforeach()

set(NR_ADAPT_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/adapt")
file(MAKE_DIRECTORY "${NR_ADAPT_GENERATED}")
foreach(pair IN ITEMS "AdaptColorCS:color" "AdaptMotionCS:motion" "ScaleOutputCS:output" "AdaptDepthCS:depth")
  string(REPLACE ":" ";" parts "${pair}")
  list(GET parts 0 entry)
  list(GET parts 1 name)
  add_custom_command(OUTPUT "${NR_ADAPT_GENERATED}/nr_adapt_${name}.h"
    COMMAND "${NR_SCALE_DXC}" -E "${entry}" -T cs_6_0 -O3 -WX -Ges -Vn "g_nr_adapt_${name}" -Fh "${NR_ADAPT_GENERATED}/nr_adapt_${name}.h" "${CMAKE_CURRENT_SOURCE_DIR}/nr_adapt.hlsl"
    DEPENDS nr_adapt.hlsl VERBATIM)
  list(APPEND NR_ADAPT_HEADERS "${NR_ADAPT_GENERATED}/nr_adapt_${name}.h")
endforeach()
add_custom_target(nr_scale_shaders DEPENDS ${NR_SCALE_SHADERS})
add_library(nr_scale_d3d12 STATIC d3d12_backend.cpp input_adapter_d3d12.cpp ${NR_ADAPT_HEADERS})
target_include_directories(nr_scale_d3d12 PRIVATE "${NR_ADAPT_GENERATED}")
target_link_libraries(nr_scale_d3d12 PUBLIC nr_scale_policy d3d12)
target_compile_definitions(nr_scale_d3d12 PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(nr_scale_d3d12 PRIVATE /W4 /WX /permissive- /EHsc)
add_dependencies(nr_scale_d3d12 nr_scale_shaders)

set(NR_SCALE_SHADERS ${NR_SCALE_SHADERS} PARENT_SCOPE)
