cmake_minimum_required(VERSION 3.20)
project(nr_recording_observer LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Recording observer requires Windows x64 MSVC ABI")
endif()
add_library(nr_recording_observer recording_observer.cpp windows_observer.cpp compute_state.cpp recording_table.cpp)
target_compile_features(nr_recording_observer PUBLIC cxx_std_17)
target_include_directories(nr_recording_observer PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(nr_recording_observer PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(nr_recording_observer PRIVATE /W4 /WX /permissive- /EHsc)
