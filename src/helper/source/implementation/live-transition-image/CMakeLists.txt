cmake_minimum_required(VERSION 3.20)
project(nr_live_image LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows MSVC required")
endif()
add_library(nr_live_image STATIC live_image.cpp)
target_compile_features(nr_live_image PUBLIC cxx_std_17)
target_include_directories(nr_live_image PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(nr_live_image PUBLIC bcrypt)
target_compile_definitions(nr_live_image PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(nr_live_image PRIVATE /W4 /WX /permissive- /EHsc)
